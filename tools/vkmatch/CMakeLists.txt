find_package(Vulkan REQUIRED)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
set(VKMATCH_SPV "${CMAKE_CURRENT_BINARY_DIR}/fpstudio-match.spv")
add_custom_command(OUTPUT "${VKMATCH_SPV}"
  COMMAND "${GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "${CMAKE_CURRENT_SOURCE_DIR}/match.comp" -o "${VKMATCH_SPV}"
  DEPENDS match.comp VERBATIM)
add_custom_target(vkmatch-shader DEPENDS "${VKMATCH_SPV}")
add_library(vkmatch STATIC vkmatch.cpp)
set_target_properties(vkmatch PROPERTIES AUTOMOC OFF AUTOUIC OFF)
target_include_directories(vkmatch PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(vkmatch PUBLIC Vulkan::Vulkan)
add_executable(fpstudio-vkmatch main.cpp)
target_compile_definitions(fpstudio-vkmatch PRIVATE VKMATCH_SHADER_PATH="${VKMATCH_SPV}")
target_link_libraries(fpstudio-vkmatch PRIVATE vkmatch Qt6::Core Qt6::Gui)
set_target_properties(fpstudio-vkmatch PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_dependencies(fpstudio-vkmatch vkmatch-shader)
add_executable(fpstudio-auth-match authmatch.cpp)
target_link_libraries(fpstudio-auth-match PRIVATE Qt6::Core Qt6::Gui)
set_target_properties(fpstudio-auth-match PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(BUILD_TESTING)
  add_executable(authpolicy-test authpolicy_test.cpp)
  target_link_libraries(authpolicy-test PRIVATE Qt6::Core)
  add_test(NAME auth-policy-fail-closed COMMAND authpolicy-test)
  add_executable(authsearch-test search_test.cpp)
  target_link_libraries(authsearch-test PRIVATE Qt6::Core Qt6::Gui)
  add_dependencies(authsearch-test fpstudio-vkmatch)
  add_test(NAME auth-search-geometric-controls COMMAND authsearch-test $<TARGET_FILE:fpstudio-vkmatch>)
  set_tests_properties(auth-search-geometric-controls PROPERTIES TIMEOUT 90)
  add_executable(minutiae-test minutiae_test.cpp)
  target_link_libraries(minutiae-test PRIVATE Qt6::Core)
  add_test(NAME minutiae-interior-controls COMMAND minutiae-test)
  add_executable(ridge-roi-test ridge_roi_test.cpp)
  target_link_libraries(ridge-roi-test PRIVATE Qt6::Core)
  add_test(NAME ridge-roi-segmentation COMMAND ridge-roi-test)
  add_executable(vkmatch-test test.cpp)
  set_target_properties(vkmatch-test PROPERTIES AUTOMOC OFF AUTOUIC OFF)
  target_link_libraries(vkmatch-test PRIVATE vkmatch)
  add_dependencies(vkmatch-test vkmatch-shader)
  add_test(NAME vulkan-cpu-parity COMMAND vkmatch-test "${VKMATCH_SPV}")
  set_tests_properties(vulkan-cpu-parity PROPERTIES TIMEOUT 120)
endif()
install(TARGETS fpstudio-vkmatch RUNTIME DESTINATION bin)
install(FILES "${VKMATCH_SPV}" DESTINATION bin)
