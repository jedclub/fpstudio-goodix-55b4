cmake_minimum_required(VERSION 3.21)
project(fpstudio_pam LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTOUIC OFF)
add_compile_options(-Wall -Wextra -Werror -fstack-protector-strong)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
add_executable(fpstudio-fprint-worker worker.c)
target_link_libraries(fpstudio-fprint-worker PRIVATE PkgConfig::SYSTEMD)
add_library(pam_fpstudio MODULE pam_fpstudio.c)
set_target_properties(pam_fpstudio PROPERTIES PREFIX "")
target_link_libraries(pam_fpstudio PRIVATE pam)
include(CTest)
if(BUILD_TESTING)
  add_executable(pam-messages-test test_messages.c)
  add_test(NAME pam-message-localisation COMMAND pam-messages-test)
  add_executable(fake-fprint-worker test_worker.c)
  add_library(pam_fpstudio_test MODULE pam_fpstudio.c)
  set_target_properties(pam_fpstudio_test PROPERTIES PREFIX "")
  target_compile_definitions(pam_fpstudio_test PRIVATE FPSTUDIO_TESTING
    FP_WORKER="$<TARGET_FILE:fake-fprint-worker>")
  target_link_libraries(pam_fpstudio_test PRIVATE pam)
  add_library(pam_test_token MODULE test_token.c)
  set_target_properties(pam_test_token PROPERTIES PREFIX "")
  target_link_libraries(pam_test_token PRIVATE pam)
  add_executable(pam-conversation-test test_conversation.c)
  find_package(Threads REQUIRED)
  target_link_libraries(pam-conversation-test PRIVATE pam pam_misc Threads::Threads)
  foreach(mode password match failure cancel unavailable timeout malformed repeat kde-match kde-unavailable kde-timeout kde-malformed)
    add_test(NAME simultaneous-${mode} COMMAND pam-conversation-test
      $<TARGET_FILE:pam_fpstudio_test> ${mode} $<TARGET_FILE:pam_test_token>)
    set_tests_properties(simultaneous-${mode} PROPERTIES TIMEOUT 10)
  endforeach()
endif()
